### 3. Concrete Test Design & Failing Assertions

```ts
it("preserves independent worktree paths, cwds, and worktree metadata across serialization and restart deserialization for multiple distinct worktrees", () => {
  const state = workspaceState();
  // 1. Configure two distinct worktrees and isolated sessions/tabs
  state.worktrees = [
    { path: "/workspace/main", branch: "main", head: "111", bare: false, detached: false, locked: null, prunable: null },
    { path: "/workspace/feat", branch: "refs/heads/orca/default/feat", head: "222", bare: false, detached: false, locked: null, prunable: null },
  ];
  state.sessions["sess-feat"] = {
    id: "sess-feat",
    cwd: "/workspace/feat",
    worktreePath: "/workspace/feat",
    workspaceId: "default",
    worktree: { wsId: "default", slug: "feat" },
    backendSessionId: "backend-2",
    lifecycle: "working",
  };
  state.layout.tabs.push({ id: "tab-2", label: "feat", sessionId: "sess-feat" });
  state.layout.layoutsByTabId["tab-2"] = {
    root: { type: "leaf", leafId: "leaf-feat" },
    activeLeafId: "leaf-feat",
    expandedLeafId: null,
    sessionIdsByLeafId: { "leaf-feat": "sess-feat" },
  };

  // 2. Serialize and restore across restart (dead backend sessions)
  const serialized = serializeWorkspaceState("default", "/workspace/main", state);
  const restored = deserializeWorkspaceState("default", serialized, new Set());

  // 3. Expected assertions verifying both worktrees survived independently
  expect(restored).not.toBeNull();
  expect(restored!.sessions["sess-1"].worktreePath).toBe("/workspace/main");
  expect(restored!.sessions["sess-1"].cwd).toBe("/workspace/main/packages/api");
  expect(restored!.sessions["sess-1"].worktree).toBeNull();

  expect(restored!.sessions["sess-feat"].worktreePath).toBe("/workspace/feat");
  expect(restored!.sessions["sess-feat"].cwd).toBe("/workspace/feat");
  expect(restored!.sessions["sess-feat"].worktree).toEqual({ wsId: "default", slug: "feat" });
  expect(restored!.sessions["sess-feat"].backendSessionId).toBeNull(); // Dead session marked for re-spawn
});
```

---

### 4. Mechanism Failure Explanation (Why Red vs. Incidental)
* **Mechanism:** When multiple worktrees exist under a single registered project workspace, serialization and deserialization must retain the distinct `worktreePath`, working directories (`cwd`), and parsed worktree slugs `{ wsId, slug }` for every terminal session rather than collapsing them to `repoRoot` / active worktree path.
* **Non-incidental assertion:** Asserting `restored.sessions["sess-feat"].worktreePath === "/workspace/feat"` and `restored.sessions["sess-feat"].worktree === { wsId: "default", slug: "feat" }` directly validates that worktree identity scoping survives independently of the main branch worktree without relying on UI timing or DOM order.
