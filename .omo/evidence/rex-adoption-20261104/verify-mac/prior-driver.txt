#!/bin/zsh
cd /Users/I552267/code/ferryx-wt/rex-buildgreen-20261104 || exit 1
git fetch origin refs/heads/ci/rex-buildgreen-25 || exit 1
git checkout -- ui/public/search.json
git checkout --detach FETCH_HEAD || exit 1
export PATH=/Users/I552267/code/ferryx-wt/.zig-0.17.0:$PATH
export ZIG=/Users/I552267/code/ferryx-wt/.zig-0.17.0/zig
export CARGO_TARGET_DIR=/Users/I552267/code/ferryx-wt/.target
export CARGO_BUILD_JOBS=4 CMAKE_POLICY_VERSION_MINIMUM=3.5
mkdir -p /tmp/rex-buildgreen-logs
bun run --cwd ui typecheck:test > /tmp/rex-buildgreen-logs/ui-typecheck.log 2>&1
echo "ui-typecheck EXIT=$?" >> /tmp/rex-buildgreen-logs/exits.log
bun scripts/fetch-ffmpeg.mjs > /tmp/rex-buildgreen-logs/ffmpeg.log 2>&1
echo "ffmpeg EXIT=$?" >> /tmp/rex-buildgreen-logs/exits.log
for mode in default headless clippy test; do
  case "$mode" in
    default) args=(check --all-targets);;
    headless) args=(check --no-default-features --all-targets);;
    clippy) args=(clippy --all-targets);;
    test) args=(test --lib --no-run);;
  esac
  if [ "$mode" = clippy ]; then
    cargo "${args[@]}" --manifest-path src-tauri/Cargo.toml --locked -- -D warnings > /tmp/rex-buildgreen-logs/$mode.log 2>&1
  else
    cargo "${args[@]}" --manifest-path src-tauri/Cargo.toml --locked > /tmp/rex-buildgreen-logs/$mode.log 2>&1
  fi
  echo "$mode EXIT=$?" >> /tmp/rex-buildgreen-logs/exits.log
done
echo DONE >> /tmp/rex-buildgreen-logs/exits.log
